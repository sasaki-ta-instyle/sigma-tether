# LLDB script: SampleAPP を走らせて SIGMA SDK 内部の PTP コマンド送信時の
# 引数・スレッド・delegate 実体を採取する (Spike 4 + 5 兼用)
#
# 使い方:
#   (lldb) source scripts/spike4_lldb_commands.txt
#   (lldb) run
#
# 想定シナリオ:
#   SampleAPP がカメラ検出→ConfigAPI→GetCamStatus2 の polling を始めると、
#   下記の breakpoint が順次 hit する。各 hit で自動的に情報を dump して continue。
#
# --------------------------------------------------------------------------
# 呼び出し規約 (Codex Nice 9 で指摘: line 44 の register が 1 個ずれていた問題を修正)
# --------------------------------------------------------------------------
# Objective-C method の実引数レイアウト:
#   [自己] [SEL] [arg1] [arg2] [arg3] [arg4] [arg5] [arg6] [arg7...]
#
# x86_64 (Rosetta 2 実行 or Intel Mac):
#   self=$rdi  _cmd=$rsi  arg1=$rdx  arg2=$rcx  arg3=$r8  arg4=$r9
#   arg5=[rsp+8 or +16 (prologue 有無に依存)]
#
# arm64 (Apple Silicon native SampleAPP):
#   self=$x0   _cmd=$x1   arg1=$x2   arg2=$x3   arg3=$x4  arg4=$x5
#   arg5=$x6   arg6=$x7   arg7=[sp+0]
#
# 本 script は x86_64 を前提に書かれている。arm64 で使う場合は
# `$rdi→$x0`, `$rsi→$x1`, `$rdx→$x2`, `$rcx→$x3`, `$r8→$x4`, `$r9→$x5`
# のように機械的に置換すること。SampleAPP を Rosetta で強制起動するには:
#   arch -x86_64 /path/to/SampleAPP.app/Contents/MacOS/SampleAPP

# ========================================================================
# Breakpoint 1: sgm_GetCamStatus2 の 7 引数版 (0x902c を発行する箇所)
# operationCode1..3 の実値を採取。
#
# selector: +sgm_GetCamStatus2:buffLength:recvLength:operationCode1:operationCode2:operationCode3:cameraHandle:
#   arg1 (outBuffer, void*)      → $rdx
#   arg2 (buffLength, UInt32)    → $rcx
#   arg3 (recvLength, UInt32*)   → $r8
#   arg4 (operationCode1, UInt32)→ $r9      ★ 採取
#   arg5 (operationCode2, UInt32)→ [rsp+16] ★ 採取
#   arg6 (operationCode3, UInt32)→ [rsp+24] ★ 採取
#   arg7 (cameraHandle, id)      → [rsp+32]
# ========================================================================
breakpoint set --name "+[sgm_GetCamStatus2 sgm_GetCamStatus2:buffLength:recvLength:operationCode1:operationCode2:operationCode3:cameraHandle:]"
breakpoint command add --script-type default
p (uint32_t)$rcx
memory read --format x --size 4 --count 1 $r8
p (uint32_t)$r9
memory read --format x --size 4 --count 3 $rsp+16
memory read --format x --size 8 --count 1 $rsp+40
bt 5
continue
DONE

# ========================================================================
# Breakpoint 2: PTP_Command:param:commandType:retry: (最内 PTP 送信 wrapper)
# param (PassThrough struct) の中身を dump
#
# selector: -PTP_Command:param:commandType:retry:
#   arg1 (camera, ICCameraDevice*) → $rdx
#   arg2 (param, PassThrough*)     → $rcx   ★ 中身を dump (80 bytes = 20 uint32)
#   arg3 (commandType, int)        → $r8    ★ 実値採取 (0/1/2 のどれか)
#   arg4 (retry, int)              → $r9    ★ 実値採取
# ========================================================================
breakpoint set --name "-[DeviceInterface PTP_Command:param:commandType:retry:]"
breakpoint command add --script-type default
po (id)$rdx
memory read --format x --size 4 --count 20 $rcx
p (int)$r8
p (int)$r9
bt 5
continue
DONE

# ========================================================================
# Breakpoint 2b: PTP_ReceiveData / PTP_SendCommand / PTP_SendData 系
# GetDeviceInfo (0x1001) がどの selector で送られるかを確認
# ========================================================================
breakpoint set --name "-[DeviceInterface PTP_ReceiveData:param:retry:]"
breakpoint command add --script-type default
po (id)$rdx
memory read --format x --size 4 --count 20 $rcx
p (int)$r8
bt 5
continue
DONE

breakpoint set --name "-[DeviceInterface PTP_SendCommand:param:retry:]"
breakpoint command add --script-type default
po (id)$rdx
memory read --format x --size 4 --count 20 $rcx
p (int)$r8
bt 5
continue
DONE

# ========================================================================
# Breakpoint 3: didSendPTPCommand callback (どのスレッドで発火するか)
#
# selector: -didSendPTPCommand:inData:response:error:contextInfo:
#   arg1 (command, NSData*)        → $rdx
#   arg2 (inData, NSData*)         → $rcx
#   arg3 (response, NSData*)       → $r8
#   arg4 (error, NSError*)         → $r9
#   arg5 (contextInfo, void*)      → [rsp+16]
# ========================================================================
breakpoint set --name "-[DeviceInterface didSendPTPCommand:inData:response:error:contextInfo:]"
breakpoint command add --script-type default
p (int)pthread_mach_thread_np(pthread_self())
p (char*)dispatch_queue_get_label((void*)0)
po (id)$rdx
po (id)$rcx
po (id)$r8
po (id)$r9
bt 5
continue
DONE

# ========================================================================
# Breakpoint 4: requestSendPTPCommand の呼び出し (sendCommandDelegate 実体を採取)
#
# selector: -requestSendPTPCommand:outData:sendCommandDelegate:didSendCommandSelector:contextInfo:
#   arg1 (command, NSData*)              → $rdx
#   arg2 (outData, NSData*)              → $rcx
#   arg3 (sendCommandDelegate, id)       → $r8    ★ delegate 実体
#   arg4 (didSendCommandSelector, SEL)   → $r9    ★ selector 名
#   arg5 (contextInfo, void*)            → [rsp+16]
# ========================================================================
breakpoint set --name "-[ICCameraDevice requestSendPTPCommand:outData:sendCommandDelegate:didSendCommandSelector:contextInfo:]"
breakpoint command add --script-type default
po (id)$rdx
p (void*)$rcx
po (id)$r8
p (char*)sel_getName((SEL)$r9)
bt 5
continue
DONE

# ========================================================================
# 全 breakpoint 一覧を表示して確認
# ========================================================================
breakpoint list
